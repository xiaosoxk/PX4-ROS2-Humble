cmake_minimum_required(VERSION 3.8)
project(safety_arbiter)
if(NOT CMAKE_CXX_STANDARD) set(CMAKE_CXX_STANDARD 17) endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(mavros_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(ai_msgs REQUIRED)
add_executable(safety_arbiter src/safety_arbiter_node.cpp)
ament_target_dependencies(safety_arbiter rclcpp mavros_msgs geometry_msgs sensor_msgs std_msgs ai_msgs)
install(TARGETS safety_arbiter DESTINATION lib/${PROJECT_NAME})
ament_package()
